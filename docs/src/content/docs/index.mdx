const client = await createDAVClient({
  serverUrl: 'https://caldav.icloud.com',
  credentials: {
    username: 'YOUR_APPLE_ID',
    password: 'YOUR_APP_SPECIFIC_PASSWORD',
  },
  authMethod: 'Basic',
  defaultAccountType: 'caldav',
});

// Discover all remote calendars
const calendars = await client.fetchCalendars();
console.log(\`Found \${calendars.length} calendars:\`, calendars.map((c) => c.displayName));

// Fetch all calendar events
const events = await client.fetchCalendarObjects({
  calendar: calendars[0],
});
console.log(\`Fetched \${events.length} calendar events.\`);`;

## Key Features

<CardGrid>
  <Card title="Universal Portability" icon="puzzle">
    Designed with zero Node-only globals (`fs`, `net`, `stream`). Works out of the box in browsers, Node.js, Cloudflare Workers, Electron, Bun, and Deno.
  </Card>

  <Card title="Delta Sync & CTAGs" icon="forward-slash">
    Native support for RFC 6578 sync-collection reports and WebDAV `getctag` headers for instant, low-bandwidth delta updates.
  </Card>

  <Card title="Real Provider Quirks Handled" icon="approve-check">
    Pre-configured normalization routines for Google's UID renames, Apple's custom namespaces, and Nextcloud's deletion quirks.
  </Card>

  <Card title="AI-Ready Documentation" icon="open-book">
    Provides structured [`/llms.txt`](/llms/) and plain-text feeds optimized for ingestion by Cursor, Claude, and custom LLM workflows.
  </Card>
</CardGrid>
